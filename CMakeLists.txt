cmake_minimum_required(VERSION 3.20)
project(NickelReconstruction LANGUAGES C CXX)
option(NICKEL_ENABLE_RENDERER_TESTS "Register own-process DX11/DX12 renderer lifecycle probes" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nickel_core STATIC src/module_core.cpp)
target_include_directories(nickel_core PUBLIC include)
set_target_properties(nickel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(nickel_module_runtime SHARED src/runtime_api.cpp)
target_link_libraries(nickel_module_runtime PRIVATE nickel_core)
if(MSVC)
  target_compile_options(nickel_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(nickel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
# On Windows the overlay, the launcher payload and most tests need the submodules; without them the build would
# silently drop those targets and tests. Other platforms build the game-independent core and its tests only.
if(WIN32)
  foreach(sub vendor/imgui/imgui.cpp vendor/minhook/CMakeLists.txt third_party/luau/CMakeLists.txt)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/${sub}")
      message(FATAL_ERROR "${sub} is missing: run git submodule update --init --recursive")
    endif()
  endforeach()
endif()
if(WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/vendor/imgui/imgui.cpp")
  enable_language(RC)
  add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
  # Luau (third_party/luau, MIT): runtime for the Scripting tab
  set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
  add_subdirectory(third_party/luau EXCLUDE_FROM_ALL)
  configure_file(native/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/overlay_resources.rc" @ONLY)
  add_library(nickel_overlay SHARED native/overlay.cpp native/menu.cpp native/media.cpp
    vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
    vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp vendor/imgui/backends/imgui_impl_dx12.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/overlay_resources.rc")
  target_include_directories(nickel_overlay PRIVATE vendor/imgui native)
  target_compile_definitions(nickel_overlay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  # /EHa: a fault caught by __try/__except also unwinds C++ objects (lock_guard locks are released). With /EHsc a lock
  # stayed held after "Rotations: fault caught" and froze the game.
  target_compile_options(nickel_overlay PRIVATE /bigobj /EHa /Zi)
  # Program database next to the DLL (not shipped): turns the "module+offset" of a logged fault into a function and line.
  target_link_options(nickel_overlay PRIVATE /DEBUG /OPT:REF /OPT:ICF)
  target_link_libraries(nickel_overlay PRIVATE minhook Luau.VM Luau.Compiler d3d11 d3d12 dxgi dcomp d3dcompiler windowscodecs ole32 windowsapp winhttp)
  set_target_properties(nickel_overlay PROPERTIES OUTPUT_NAME "Nickel.Overlay")
  if(MSVC)
    set_target_properties(nickel_overlay minhook PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_executable(nickel_overlay_probe native/probe.cpp)
  target_compile_definitions(nickel_overlay_probe PRIVATE NOMINMAX UNICODE _UNICODE)
  target_link_libraries(nickel_overlay_probe PRIVATE d3d11 d3d12 dxgi windowscodecs ole32)
  set_target_properties(nickel_overlay_probe PROPERTIES OUTPUT_NAME "Nickel.OverlayProbe")
  add_executable(nickel_fault_boundary_checks tests/fault_boundary_checks.cpp)
  target_include_directories(nickel_fault_boundary_checks PRIVATE native)
  target_compile_options(nickel_fault_boundary_checks PRIVATE /EHa)
  # Path Finder search on a synthetic world (the real game_movement_e.hpp, no game memory read)
  add_executable(nickel_path_finder_checks tests/path_finder_checks.cpp)
  target_include_directories(nickel_path_finder_checks PRIVATE vendor/imgui native)
  target_compile_definitions(nickel_path_finder_checks PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(nickel_path_finder_checks PRIVATE /bigobj /EHa)
  target_link_libraries(nickel_path_finder_checks PRIVATE minhook windowscodecs ole32)
  add_executable(nickel_script_values_checks tests/script_values_checks.cpp)
  target_include_directories(nickel_script_values_checks PRIVATE native)
  target_link_libraries(nickel_script_values_checks PRIVATE Luau.VM)
  if(MSVC)
    set_target_properties(nickel_script_values_checks PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
add_executable(nickel_core_checks tests/module_core_checks.cpp)
target_link_libraries(nickel_core_checks PRIVATE nickel_core)
enable_testing()
add_test(NAME recovered_behavior_checks COMMAND nickel_core_checks)

  add_test(NAME tracked_inputs_checks COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/check_tracked_inputs.py)
  add_test(NAME game_builds_in_step COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/check_game_builds.py)
endif()
